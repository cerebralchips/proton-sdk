cmake_minimum_required(VERSION 3.26)
option(PROTON_DDR "Place packed model weights in external memory" OFF)
project(proton_sdk C CXX ASM)
include(CheckCSourceCompiles)
# The existing lab compiler wrapper supplies Newlib link flags on compile calls.
add_compile_options(-Wno-unused-command-line-argument)
set(IREE_BUILD_COMPILER OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_BINDINGS_TFLITE OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_BINDINGS_TFLITE_JAVA OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_THREADING OFF CACHE BOOL "" FORCE)
set(IREE_SYNCHRONIZATION_DISABLE_UNSAFE ON CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_LOCAL_SYNC ON CACHE BOOL "" FORCE)
set(IREE_HAL_EXECUTABLE_LOADER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_HAL_EXECUTABLE_PLUGIN_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(IREE_HOST_BIN_DIR "${PROTON_WORK}/host-tools" CACHE PATH "" FORCE)
add_subdirectory("${PROTON_WORK}/deps/iree" "${CMAKE_BINARY_DIR}/iree")
# UART evidence uses integers and exact FP32 bit patterns; omit unsupported
# long-double formatting from the minimal console dependency.
target_compile_definitions(printf_printf PRIVATE PRINTF_SUPPORT_DECIMAL_SPECIFIERS=0 PRINTF_SUPPORT_EXPONENTIAL_SPECIFIERS=0)
add_executable(smoke runtime/proton/start.S runtime/proton/platform.c runtime/proton/console.c runtime/iree_runner.c
  "${PROTON_WORK}/generated/smoke/embed.c" "${PROTON_WORK}/generated/smoke/dispatch.o")
target_include_directories(smoke PRIVATE runtime "${PROTON_WORK}/generated/smoke")
target_link_libraries(smoke PRIVATE iree::runtime iree::hal::drivers::local_sync::sync_driver
  iree::hal::local::loaders::static_library_loader printf::printf m)
target_link_options(smoke PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/smoke.map")
if(EXISTS "${PROTON_WORK}/generated/model/model_data.c")
  set(_model_data "${PROTON_WORK}/generated/model/model_data.c")
  if(PROTON_DDR)
    set(_model_data "${PROTON_WORK}/generated/model/model_data_ddr.c")
  endif()
  add_library(proton_kernels STATIC kernels/linear.c kernels/model_ops.c "${_model_data}")
  target_include_directories(proton_kernels PUBLIC kernels runtime "${PROTON_WORK}/generated/model")
  target_compile_options(proton_kernels PRIVATE -O3)
  add_executable(kernels tests/kernels.c runtime/proton/start.S runtime/proton/platform.c runtime/proton/console.c)
  target_include_directories(kernels PRIVATE "${PROTON_WORK}/generated/tests")
  target_link_libraries(kernels PRIVATE proton_kernels printf::printf m)
  target_link_options(kernels PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/kernels.map")
  add_executable(kernels_bad tests/kernels.c runtime/proton/start.S runtime/proton/platform.c runtime/proton/console.c)
  target_include_directories(kernels_bad PRIVATE "${PROTON_WORK}/generated/tests")
  target_compile_definitions(kernels_bad PRIVATE PROTON_INJECT_FAILURE=1)
  target_link_libraries(kernels_bad PRIVATE proton_kernels printf::printf m)
  add_executable(kernel_tile tests/kernels.c runtime/proton/start.S runtime/proton/platform.c runtime/proton/console.c)
  target_include_directories(kernel_tile PRIVATE "${PROTON_WORK}/generated/tests")
  target_compile_definitions(kernel_tile PRIVATE PROTON_TILE_ONLY=1)
  target_link_libraries(kernel_tile PRIVATE proton_kernels printf::printf m)
  if(EXISTS "${PROTON_WORK}/generated/model/dispatch.o")
    foreach(variant model_step model_matrix model_scalar model_scalar_step)
      add_executable(${variant} runtime/proton/start.S runtime/proton/platform.c runtime/proton/console.c
        runtime/iree_runner.c "${PROTON_WORK}/generated/model/embed.c" "${PROTON_WORK}/generated/model/dispatch.o")
      if(PROTON_DDR)
        target_sources(${variant} PRIVATE runtime/proton/ddr_entry.c)
        target_link_options(${variant} PRIVATE "-Wl,--wrap=main")
      endif()
      target_include_directories(${variant} PRIVATE "${PROTON_WORK}/generated/model")
      target_compile_definitions(${variant} PRIVATE PROTON_MODEL=1)
      if(variant STREQUAL "model_step" OR variant STREQUAL "model_scalar_step")
        target_compile_definitions(${variant} PRIVATE PROTON_STEPS=1)
      endif()
      if(variant STREQUAL "model_scalar" OR variant STREQUAL "model_scalar_step")
        target_compile_definitions(${variant} PRIVATE PROTON_BACKEND=0)
      endif()
      target_link_libraries(${variant} PRIVATE proton_kernels iree::runtime iree::hal::drivers::local_sync::sync_driver
        iree::hal::local::loaders::static_library_loader printf::printf m)
      target_link_options(${variant} PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/${variant}.map")
    endforeach()
  endif()
endif()
